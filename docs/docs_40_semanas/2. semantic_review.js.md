const fs = require('fs');
const semana = process.argv.find(a => a.startsWith("--semana="))?.split("=")[1];
if (!semana) {
console.error("Informe --semana");
process.exit(1);
}
const inputPath = weeks/semana_${semana}.raw.md;
const outputPath = weeks/semana_${semana}.review.md;
if (!fs.existsSync(inputPath)) {
console.error("Arquivo não encontrado");
process.exit(1);
}
const content = fs.readFileSync(inputPath, 'utf-8');
const prompt = `
Você é um especialista em conteúdo para gestantes.
Melhore o texto abaixo:
REGRAS:
• NÃO mudar estrutura
• NÃO mudar quantidade de cards
• NÃO mudar tipos
• Melhorar clareza, fluidez e tom acolhedor
• Remover repetições
Retorne o conteúdo completo corrigido:
${content}
`;
fs.writeFileSync("tmp_prompt.txt", prompt);
console.log("🧠 Copie o conteúdo de tmp_prompt.txt e cole no Claude.");
console.log("Depois salve a resposta em:");
console.log(outputPath);
