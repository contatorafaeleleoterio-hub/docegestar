const fs = require('fs');
const semana = process.argv.find(a => a.startsWith("--semana="))?.split("=")[1];
if (!semana) {
console.error("Informe --semana");
process.exit(1);
}
const path = weeks/semana_${semana}.review.md;
if (!fs.existsSync(path)) {
console.error("Arquivo não encontrado");
process.exit(1);
}
const content = fs.readFileSync(path, 'utf-8');
let score = 100;
// Penalidades simples
if (content.length < 2000) score -= 15;
const repeticoes = (content.match(/bebê/g) || []).length;
if (repeticoes > 20) score -= 10;
const perguntas = (content.match(/?/g) || []).length;
if (perguntas < 3) score -= 10;
if (!content.includes("💛") && !content.includes("🤰")) {
score -= 10;
}
console.log(Score: ${score});
if (score < 85) {
console.log("❌ Reprovado");
process.exit(1);
} else {
console.log("✅ Aprovado");
}
