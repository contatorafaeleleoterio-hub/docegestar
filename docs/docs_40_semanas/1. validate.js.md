const fs = require('fs');
const TIPOS_VALIDOS = [
"HERO","STAT","LISTA","PERGUNTA","LISTA-ALERT","CHECKLIST"
];
function erro(msg) {
console.error("❌ " + msg);
process.exit(1);
}
function ok(msg) {
console.log("✅ " + msg);
}
const semana = process.argv.find(a => a.startsWith("--semana="))?.split("=")[1];
const input = process.argv.find(a => a.startsWith("--input="))?.split("=")[1] || "final";
if (!semana) erro("Informe --semana");
const path = input === "raw"
? weeks/semana_${semana}.raw.md
: weeks/semana_${semana}.md;
if (!fs.existsSync(path)) erro("Arquivo não encontrado");
const content = fs.readFileSync(path, 'utf-8');
// Conta cards por marcador ##
const cards = content.split("\n## ").slice(1);
if (cards.length !== 15) erro(Deve ter 15 cards, encontrado: ${cards.length});
cards.forEach((card, i) => {
const pos = i + 1;
const tipoMatch = card.match(/[(.*?)]/);
if (!tipoMatch) erro(Card ${pos} sem tipo);
const tipo = tipoMatch[1];
if (!TIPOS_VALIDOS.includes(tipo)) {
erro(Card ${pos} tipo inválido: ${tipo});
}
});
ok(Semana ${semana} válida);
